def max_loading(weights, cap):
    best_w = 0
    heap = [( -sum(weights), 0, 0 )]   # 上界 第0个物品 目前已装重量为0
    while heap:
        currend_bound, i, currend_w  = heappop(heap)
        if -currend_bound <= best_w:
            continue  # 剪枝 如果理论上限连历史最高记录都比不过，直接放弃

        if i == len(weights): continue # 终止条件：防止越界

        if weights[i] + currend_w <= cap:
            best_w = max(best_w, weights[i] + currend_w)
            new_bound = -(weights[i] + currend_w + sum(weights[i+1:]))
            heappush(heap,( new_bound, i+1,  weights[i] + currend_w))

        skip_bound = -(currend_w + sum(weights[i+1:]))
        heappush(heap,( skip_bound, i+1,  currend_w))

    return best_w

def sub_sum(arr, t):
    n = len(arr)
    ans = []

    def dfs(i, path, total):
        if total == t:
            ans.append(path.copy())
            return
        if i==n or total > t: return  # 剪枝

        path.append(i)
        dfs(i+1, path, total + arr[i])
        path.pop()  # 回溯
        dfs(i+1, path, total)

    dfs(0, [], 0)
    return ans

    wpl = 0
    while len(heap) > 1:
        w1 = heapq.heappop(heap)
        w2 = heapq.heappop(heap)
        wpl += (w1 + w2)
        create_node(w1, w2)  # 创建父节点 并合并节点的方法

def lcs(arr1, arr2):
    n, m = len(arr1), len(arr2)
    dp = [[0] * (n+1) for _ in range(m+1)]

    for i in range(1, m+1):
        for j in range(1, n+1):
            if arr1[j-1] == arr2[i-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])

    return dp[m][n]

def max_loading(weights, cap):
    n = len(weights)
    best = 0
    heap = [(-sum(weights), 0, 0)]  # 最小堆模拟优先队列
    while heap:
        curr_bound, i, w = heapq.heappop(heap)  # 上界，选到第i个物品的时候，已经有的重量
        if -curr_bound < best or i == n:   # 剪枝
            continue

        if weights[i] + w <= cap:  # 选择装第i个物品
            currend_w = weights[i] + w
            best = max(best, currend_w)
            new_bound = -(sum(weights[i+1:]) + currend_w)
            heapq.heappush((new_bound, i+1, currend_w))

        # 不选这个物品
        skip_bound = -(sum(weights[i+1:]) + w)
        heapq.heappush((skip_bound, i+1, w))
    return best

    n = len(weights)
    dp = [[0]*(cap+1) for _ in range(n+1)]
    for i in range(1, n+1):
        for j in range(1, cap+1):
            w = weights[i-1]
            v = values[i-1]
            if w > j:  # 如果重量大于现在的容量 就使用考虑上一个物品的dp值
                dp[i][j] = dp[i-1][j]
            elif w<=j:
                dp[i][j] = max( dp[i-1][j], dp[i-1][j-w]+v )

    return dp[n][cap]


def heap_bag(weights, values, cap):
    n = len(weights)
    best = 0
    heap = [( -sum(values), 0, 0, 0)]  # 上界 第i个物品 目前总重量 目前总价值

    while heap:
        curr_bound, i, w, v = heapq.heappop(heap)
        if -curr_bound < best or i==n:  # 剪枝
            continue

        if w + weights[i] <= cap:  # 选择这个物品
            curr_w = w + weights[i]   # 选了这个物品之后的重量
            curr_v = v + values[i]
            best = max(best, curr_v)
            new_bound = -(sum(values[i+1:]) + curr_v)
            heapq.heappush(heap, (new_bound, i+1, curr_w, curr_v))

        # 不选这个物品
        skip_bound = -(sum(values[i+1:]) + v)
        heapq.heappush(heap,(skip_bound, i+1, w, v))

    return best
